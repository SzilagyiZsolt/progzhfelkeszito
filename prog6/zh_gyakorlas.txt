//1. feladat
//Írjon függvényt amelynek két paramétere van a visszatérési értéke, pedig a két szám lkkt-je.
int lnko(int a, int b) {
    while (b != 0) {
        int temp = b;
        b = a % b;
        a = temp;
    }
    return a;
}

int lkkt(int a, int b) {
    if (a == 0 || b == 0) return 0;
    return abs(a / lnko(a, b)) * b;
}
//2. feladat
//Írjon függvényt amelynek egy 3x3-as tömb kétdimenziós tömb a paramétere és visszaadja a soronkénti maximum és minimum különbségét egy
//3 méretű tömbben.
vector<int> soronkentiKulonbseg(int tomb[3][3]) {
    vector<int> eredmeny(3);

    for (int i = 0; i < 3; i++) {
        int sorMax = tomb[i][0];
        int sorMin = tomb[i][0];

        for (int j = 1; j < 3; j++) {
            if (tomb[i][j] > sorMax) sorMax = tomb[i][j];
            if (tomb[i][j] < sorMin) sorMin = tomb[i][j];
        }

        eredmeny[i] = sorMax - sorMin;
    }

    return eredmeny;
}

//3. feladat
//Írjon függvényt amelynek 3 integer paramétere van(a,b,c). A legkiseb és a legnagyobb értéket írja felül 0-ra, majd cserélje meg az (a) és (c), majd a (b) és (a)
//változók értékét. A függvény visszatérési értéke void. (a<c, b>a). A cserék utáni állapotot a main függvényben írja ki.
void muvelet(int &a, int &b, int &c) {
    // 1. Megkeressük az eredeti legkisebb és legnagyobb értéket
    int legkisebb = min({a, b, c});
    int legnagyobb = max({a, b, c});

    // Segédváltozók az eredeti értékek megőrzéséhez a nullázás idejére
    int origA = a, origB = b, origC = c;

    // 2. A legkisebb és legnagyobb értékeket felülírjuk 0-ra
    if (origA == legkisebb || origA == legnagyobb) a = 0;
    if (origB == legkisebb || origB == legnagyobb) b = 0;
    if (origC == legkisebb || origC == legnagyobb) c = 0;

    // 3. Megcseréljük az (a) és (c) értékét
    int temp = a;
    a = c;
    c = temp;

    // 4. Megcseréljük a (b) és (a) értékét
    temp = b;
    b = a;
    a = temp;
}

    ifstream bemenet("sorok.txt");
    if (!bemenet.is_open()) {
        cerr << "Hiba: A sorok.txt nem talalhato!" << endl;

    }
